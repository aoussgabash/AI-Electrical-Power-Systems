# Import datetime so the agent can read the current computer time.
# استيراد datetime كي يتمكن الوكيل من قراءة الوقت الحالي من الحاسوب.
from datetime import datetime


# Define the calculator tool. A function is a reusable group of instructions.
# تعريف أداة الحاسبة. الدالة هي مجموعة تعليمات يمكن استخدامها عدة مرات.
def calculator(expression):
    # Try to evaluate the mathematical expression entered by the user.
    # محاولة حساب العملية الرياضية التي أدخلها المستخدم.
    try:
        # Return sends the calculated value back to the agent.
        # تعيد return القيمة المحسوبة إلى الوكيل.
        return eval(expression, {"__builtins__": {}})

    # If an error occurs, do not stop the program; return a clear message.
    # عند حدوث خطأ لا يتوقف البرنامج، بل يعيد رسالة واضحة.
    except Exception:
        return "Calculation failed | تعذر إجراء العملية الحسابية"


# Define the second tool, which reads the current time.
# تعريف الأداة الثانية التي تقرأ الوقت الحالي.
def get_time():
    # %H means hours and %M means minutes.
    # تعني %H الساعات وتعني %M الدقائق.
    return datetime.now().strftime("%H:%M")


# Define the agent's decision-making function.
# تعريف الدالة المسؤولة عن اتخاذ القرار داخل الوكيل.
def simple_agent(user_message):
    # lower() normalizes letters; strip() removes extra spaces.
    # توحّد lower() شكل الأحرف وتحذف strip() الفراغات الزائدة.
    message = user_message.lower().strip()

    # Decision 1: if the request starts with "احسب", use the calculator.
    # القرار 1: إذا بدأ الطلب بكلمة "احسب" فاستخدم الحاسبة.
    if message.startswith("احسب"):
        # Remove the command word and keep only the mathematical expression.
        # حذف كلمة الأمر والإبقاء على العملية الرياضية فقط.
        expression = message.replace("احسب", "").strip()

        # Call the calculator tool and store its result.
        # استدعاء أداة الحاسبة وتخزين نتيجتها.
        result = calculator(expression)

        # f before the string allows us to insert the result inside the sentence.
        # تسمح f قبل النص بوضع النتيجة داخل الجملة.
        return f"نتيجة الحساب: {result}"

    # Decision 2: elif means "otherwise, check another condition".
    # القرار 2: تعني elif "وإلا افحص شرطًا آخر".
    elif "الوقت" in message or "الساعة" in message:
        # Use the time tool and place its answer in the response.
        # استخدام أداة الوقت ووضع إجابتها داخل الرد.
        return f"الوقت الآن هو: {get_time()}"

    # Decision 3: respond to a greeting without using another tool.
    # القرار 3: الرد على التحية دون استخدام أداة أخرى.
    elif "مرحبا" in message or "السلام" in message:
        return "مرحبًا، أستطيع الحساب وإخبارك بالوقت."

    # If no previous condition matches, provide examples the user can try.
    # إذا لم يتحقق أي شرط سابق، يعرض الوكيل أمثلة يمكن للمستخدم تجربتها.
    else:
        return "جرّب: احسب 5 * 6 أو ما الوقت؟"


# Start an endless conversation loop. It repeats until break is executed.
# بدء حلقة محادثة مستمرة تتكرر حتى تنفيذ break.
while True:
    # input() waits for the student to type a message.
    # تنتظر input() حتى يكتب الطالب رسالة.
    user_input = input("الطالب: ")

    # Stop the loop when the student writes "خروج".
    # إيقاف الحلقة عندما يكتب الطالب كلمة "خروج".
    if user_input.lower() == "خروج":
        print("الوكيل: إلى اللقاء.")
        break

    # Send the student's message to the agent's decision function.
    # إرسال رسالة الطالب إلى دالة اتخاذ القرار.
    answer = simple_agent(user_input)

    # Display the final answer on the screen.
    # عرض الإجابة النهائية على الشاشة.
    print("الوكيل:", answer)
Teaching and safety note | ملاحظة تعليمية وأمنية:
eval is used only to simplify this first lesson. A real application should use a safe mathematical parser and should never execute unrestricted user input.
استُخدمت eval فقط لتبسيط هذا الدرس الأول. يجب في التطبيق الحقيقي استخدام محلل رياضي آمن وعدم تنفيذ مدخلات المستخدم دون قيود.

Why is it an agent? | لماذا هو وكيل؟

It receives a goal, makes a decision, selects an action, uses a tool, and returns a result.

لأنه يستقبل هدفًا، ويتخذ قرارًا، ويختار فعلًا، ويستخدم أداة، ثم يعيد النتيجة.

Is it real AI yet? | هل هو ذكاء اصطناعي حقيقي؟

It is a beginner rule-based agent. Later, the fixed conditions can be replaced by a language model.

إنه وكيل تعليمي قائم على القواعد. يمكن لاحقًا استبدال الشروط الثابتة بنموذج لغوي.
